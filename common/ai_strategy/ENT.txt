ENT_prepare_for_intervention_against_WCA = {
	allowed = {
		is_in_faction_with = ENT
		NOT = { tag = CAN }
		is_american_tag = no
	}

	enable = {
		NOT = { has_global_flag = USA_civil_war_over_flag }
		OR = {
			has_war_with = WCA

				is_in_faction_with = CAN
				check_variable = { CAN.ai_strategy_prepare_for_war@WCA > 0 }
			}
			AND = {
				is_in_faction_with = NEE

			}
		}

		type = force_defend_ally_borders
		id = CAN
		value = 100
	}

		type = force_defend_ally_borders
		id = NEE
		value = 100
	}
	ai_strategy = { type = area_priority id = north_america		value = 50 }
	ai_strategy = {
		type = front_unit_request
		tag = WCA
		value = 60
	}
	ai_strategy = {
		type = prepare_for_war
		id = WCA
		value = 400
	}
	ai_strategy = {
		type = conquer
		id = WCA
		value = 400 #counters avoid_starting_wars
	}
	ai_strategy = {
		type = invade
		id = WCA
		value = -500
	}

	ai_strategy = {
		type = naval_dominance
		id = usa_east_seas

ENT_sync_invasions = {
	allowed = {
		is_in_faction_with = ENT
	}
	enable = {
		has_war_with = INT
		OR = {
			FRA = {
				exists = yes
				has_capitulated = no
			}
			ENG = {
				exists = yes
				has_capitulated = no
			}
		}

		#activate every 60 days
		set_temp_variable = { days = global.num_days }
		modulo_temp_variable = { days = 90 }
		check_variable = { days < 60 }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		area = western_europe
		ordertype = invasion
		execute_order = no
	}
}
