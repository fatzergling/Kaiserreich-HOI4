CHI_WCA_joint_attack_on_japan = {
	allowed = {
		OR = {
			original_tag = CHI
			original_tag = WCA
		}
	}
	enable = {
		has_wargoal_against = {
			target = JAP
			type = topple_government
		}
		surrender_progress < 0.05
		any_country_of = {
			target = { CHI WCA }
			has_war_with = JAP
		}
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = conquer
		id = JAP
		value = 200
	}
	ai_strategy = {
		type = declare_war
		id = JAP
		value = 500 #strategy >= 500 forces declaration
	}
}

CHI_attack_nanjing = {
	allowed = {
		original_tag = CHI
	}
	enable = {
		has_war_with = LEP
		#activate every  5 days, then pause for another 5
		set_temp_variable = { frequency = global.num_days }
		modulo_temp_variable = { frequency = 50 }
		check_variable = { frequency < 45 }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		tag = LEP
		priority = 200
		ordertype = front
		execution_type = rush_weak
		execute_order = yes
	}
	ai_strategy = {
		type = conquer
		id = LEP
		value = 100
	}
	ai_strategy = {
		type = garrison
		value = -2000
	}
}
